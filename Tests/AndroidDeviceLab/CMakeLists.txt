cmake_minimum_required(VERSION 3.30)
project(OpenRHINativeLab LANGUAGES C CXX)
get_filename_component(OPENRHI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(OPENRHI_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(OPENRHI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OPENRHI_ENABLE_NONE_SUPPORT ON CACHE BOOL "" FORCE)
set(OPENRHI_ENABLE_NIS_SDK ON CACHE BOOL "" FORCE)
set(OPENRHI_ENABLE_ANF_SDK ON CACHE BOOL "" FORCE)
set(OPENRHI_ANF_SDK_ROOT "${ANF_SDK_ROOT}" CACHE PATH "" FORCE)
set(OPENRHI_ENABLE_NVTX_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENRHI_ROOT}" openrhi)

set(qualification "${OPENRHI_ROOT}/Tests/DeviceQualification.cpp")
set(smoke "${OPENRHI_ROOT}/Tests/BackendSmoke.cpp")
set(policy "${OPENRHI_ROOT}/Tests/DevicePolicyTests.cpp")
set(public_api "${OPENRHI_ROOT}/Tests/PublicApiTests.cpp")
set(anf_probe "${CMAKE_CURRENT_SOURCE_DIR}/CompatibilityProbe.cpp")
set_source_files_properties("${qualification}" PROPERTIES COMPILE_DEFINITIONS main=OpenRHIQualificationMain)
set_source_files_properties("${smoke}" PROPERTIES COMPILE_DEFINITIONS main=OpenRHISmokeMain)
set_source_files_properties("${policy}" PROPERTIES COMPILE_DEFINITIONS main=OpenRHIPolicyMain)
set_source_files_properties("${public_api}" PROPERTIES COMPILE_DEFINITIONS main=OpenRHIPublicMain)
set_source_files_properties("${anf_probe}" PROPERTIES COMPILE_DEFINITIONS main=ANFCompatibilityMain)
add_library(OpenRHINativeLab SHARED NativeTests.cpp AnfDispatch.cpp "${qualification}" "${smoke}" "${policy}" "${public_api}"
    "${OPENRHI_ROOT}/Tests/PublicFacade.c" "${anf_probe}")
target_compile_features(OpenRHINativeLab PRIVATE cxx_std_17)
target_include_directories(OpenRHINativeLab PRIVATE "${OPENRHI_ROOT}/Source/Shared" "${ANF_SDK_ROOT}/public" "${FETCHCONTENT_SOURCE_DIR_VULKAN_HEADERS}/include")
target_compile_definitions(OpenRHINativeLab PRIVATE OPENRHI_TEST_NONE=1 LAB_CONFIGURATION="${CMAKE_BUILD_TYPE}")
target_link_libraries(OpenRHINativeLab PRIVATE OpenRHI android log vulkan dl)
target_link_options(OpenRHINativeLab PRIVATE -Wl,-z,max-page-size=16384)

openrhi_compile_shader(NAME DescriptorCompute SOURCE "${OPENRHI_ROOT}/Tests/DescriptorCompute.slang" ENTRY descriptorMain STAGE compute
    TARGET spirv INCLUDES "${OPENRHI_ROOT}/Include" OUTPUT compute)
openrhi_compile_shader(NAME IndirectDraw.vs SOURCE "${OPENRHI_ROOT}/Tests/IndirectDraw.slang" ENTRY vsMain STAGE vertex TARGET spirv OUTPUT vertex)
openrhi_compile_shader(NAME IndirectDraw.ps SOURCE "${OPENRHI_ROOT}/Tests/IndirectDraw.slang" ENTRY psMain STAGE fragment TARGET spirv OUTPUT fragment)
openrhi_compile_shader(NAME MeshDraw SOURCE "${OPENRHI_ROOT}/Tests/MeshDraw.slang" ENTRY meshMain STAGE mesh TARGET spirv OUTPUT mesh)
openrhi_compile_shader(NAME TaskDraw SOURCE "${OPENRHI_ROOT}/Tests/MeshDraw.slang" ENTRY taskMain STAGE amplification TARGET spirv OUTPUT task)
openrhi_compile_shader(NAME CacheProbe SOURCE CacheProbe.slang ENTRY cacheMain STAGE compute TARGET spirv OUTPUT cache)
add_custom_target(OpenRHILabShaders DEPENDS "${compute}" "${vertex}" "${fragment}" "${mesh}" "${task}" "${cache}")
add_dependencies(OpenRHINativeLab OpenRHILabShaders)
