cmake_minimum_required(VERSION 3.30)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OpenRHIBindlessMaterialFixture LANGUAGES CXX)
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Qualified fixture configurations" FORCE)
    enable_testing()
    set(OPENRHI_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../_OpenRHI_SDK" CACHE PATH "Qualified OpenRHI SDK")
    add_library(OpenRHI SHARED IMPORTED)
    foreach(configuration IN ITEMS Debug Release)
        string(TOUPPER "${configuration}" upper)
        set_target_properties(OpenRHI PROPERTIES
            IMPORTED_IMPLIB_${upper} "${OPENRHI_SDK}/Lib/${configuration}/OpenRHI.lib"
            IMPORTED_LOCATION_${upper} "${OPENRHI_SDK}/Lib/${configuration}/OpenRHI.dll")
    endforeach()
    set_target_properties(OpenRHI PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${OPENRHI_SDK}/Include")
    include("${OPENRHI_SDK}/CMake/OpenRHISlang.cmake")
endif()
if(NOT WIN32)
    message(FATAL_ERROR "This native fixture qualifies the D3D12 descriptor-heap transport on Windows")
endif()
add_executable(OpenRHIBindlessMaterial Main.cpp Reference.h)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set_target_properties(OpenRHIBindlessMaterial PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<TARGET_FILE_DIR:OpenRHI>")
endif()
target_link_libraries(OpenRHIBindlessMaterial PRIVATE OpenRHI)
target_compile_features(OpenRHIBindlessMaterial PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(OpenRHIBindlessMaterial PRIVATE /W4 /WX)
endif()
openrhi_compile_shader(NAME BindlessMaterialHeap SOURCE Material.slang ENTRY materialMain STAGE compute TARGET dxil
    DEFINES OPENRHI_MATERIAL_CONVENTIONAL=0 OUTPUT material_heap)
openrhi_compile_shader(NAME BindlessMaterialConventional SOURCE Material.slang ENTRY materialMain STAGE compute TARGET dxil
    DEFINES OPENRHI_MATERIAL_CONVENTIONAL=1 OUTPUT material_conventional)
add_custom_target(OpenRHIBindlessMaterialShaders DEPENDS "${material_heap}" "${material_conventional}")
add_dependencies(OpenRHIBindlessMaterial OpenRHIBindlessMaterialShaders)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    # Isolated runtime copies preserve the production SDK and library outputs
    file(GLOB sdk_runtime_dlls "${OPENRHI_SDK}/Lib/Release/*.dll")
    foreach(runtime IN LISTS sdk_runtime_dlls)
        get_filename_component(name "${runtime}" NAME)
        add_custom_command(TARGET OpenRHIBindlessMaterial POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${OPENRHI_SDK}/Lib/$<CONFIG>/${name}" "$<TARGET_FILE_DIR:OpenRHIBindlessMaterial>/${name}")
    endforeach()
    add_custom_command(TARGET OpenRHIBindlessMaterial POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${OPENRHI_SDK}/Lib/$<CONFIG>/AgilitySDK" "$<TARGET_FILE_DIR:OpenRHIBindlessMaterial>/AgilitySDK")
endif()
add_test(NAME OpenRHI.BindlessMaterial.Reference COMMAND OpenRHIBindlessMaterial --cpu-only)
add_test(NAME OpenRHI.D3D12.BindlessMaterial COMMAND OpenRHIBindlessMaterial --native "${material_heap}" "${material_conventional}"
    "${CMAKE_BINARY_DIR}/Testing/BindlessMaterial/$<CONFIG>/native")
set_tests_properties(OpenRHI.D3D12.BindlessMaterial PROPERTIES TIMEOUT 120)
add_test(NAME OpenRHI.D3D12.BindlessMaterialAnisotropy16 COMMAND OpenRHIBindlessMaterial --native-aniso16 "${material_heap}" "${material_conventional}"
    "${CMAKE_BINARY_DIR}/Testing/BindlessMaterial/$<CONFIG>/anisotropy16")
set_tests_properties(OpenRHI.D3D12.BindlessMaterialAnisotropy16 PROPERTIES TIMEOUT 120)
